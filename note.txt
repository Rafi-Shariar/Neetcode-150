#include <vector>
#include <unordered_map>
#include <queue>

using namespace std;

vector<int> topKFrequent(vector<int>& nums, int k) {
    // Step 1: Count frequencies using an unordered_map (faster than map)
    unordered_map<int, int> mp;
    for (int num : nums) {
        mp[num]++;
    }

    // Step 2: Use a max-heap to store pairs of {frequency, element}
    // C++ priority_queue sorts by the first element of the pair by default
    priority_queue<pair<int, int>> pq;
    for (auto& it : mp) {
        pq.push({it.second, it.first});
    }

    // Step 3: Extract the top k elements from the heap
    vector<int> result;
    for (int i = 0; i < k; i++) {
        result.push_back(pq.top().second);
        pq.pop();
    }

    return result;
}
